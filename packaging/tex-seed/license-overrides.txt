# Licences for seed files the TeX Live 2022 database does not describe (scripts/tex-seed.py).
# One line each: file <bundle file name> <licence>... | glob <name pattern> <licence>... | package <name> <licence>...
# A file line wins over a glob line (the longest matching pattern wins), which wins over the
# package's licence in the database.
# Licence ids are the keys of TEXTS in scripts/tex-seed.py. Each entry says where the
# licence comes from; "in-file" means the file's own header states the terms and the
# file is shipped with that header unmodified.

# hyph-utf8 pattern files: each states its licence (MIT, LPPL, GPL 2 or later, LGPL, BSD,
# MPL among alternatives, or a plain all-permissive notice such as hyph-en-us.tex and
# hyph-nb/nn/no.tex) in its "licence:" header; the database has no licence for most
# hyphen-* packages. hyph-hy.tex says "LGPL" with a link to LGPL 3.0, whose text is not
# shipped (maintainer review).
glob hyph-*.tex in-file
# Loaders: "Copyright 2008-2021 TeX Users Group. You may freely use, modify and/or distribute
# this file." (header).
glob loadhyph-*.tex in-file
# Apostrophe variants of the patterns, generated by hyph-utf8 from hyph-<language>.tex (which
# the seed also holds); they have no header of their own.
glob hyph-quote-*.tex derived
# Their "licence:" header says "[None]" (hyph-ro.tex) and "NONE" (hyph-mn-cyrl-x-lmc.tex).
# Tectonic's format loads every language, so they cannot be left out (maintainer review).
file hyph-ro.tex unstated
file hyph-mn-cyrl-x-lmc.tex unstated
# Werner Lemberg's dehyph-exptl: "You may freely use, modify and/or distribute this file."
glob dehyph?-x-*.tex in-file
glob dehyph?-x-*.pat in-file

# Plain TeX's hyphen.tex: "Unlimited copying and redistribution of this file are permitted
# as long as this file is not modified" (header).
file hyphen.tex in-file
# Bernard Gaulle's placeholder files (hyphen-base): a comment, and two dummy patterns or
# none; no licence statement.
file dumyhyph.tex unstated
file zerohyph.tex unstated
# language.dat: "Created long ago by Sebastian Rahtz and others. Public domain." (header).
file language.dat pd
# Greek patterns for Ibycus encoding, translated from hyph-grc (LPPL, see hyph-grc.tex).
file ibyhyph.tex lppl1.3c

# Adobe Glyph List 2.0: Apache-2.0 (header). pdfglyphlist.txt has no licence statement;
# its header says it is compiled from the Adobe Glyph List 2.0 and the PDF 1.5 reference.
file glyphlist.txt apache2
file pdfglyphlist.txt apache2
# lcdf-typetools texglyphlist.txt: the header names no licence; lcdf-typetools, which it
# comes from, is GPL 2 or later (its COPYING).
file texglyphlist.txt gpl2+
# Generated by updmap ("maintained by updmap[-sys]", header) from the map files of the font
# packages in the bundle: lines of font names, encodings and font file names.
file pdftex.map updmap
file kanjix.map updmap
# Two TeX Live packages besides the one in the bundle ship a file of the same name; the
# headers say which: hyphen.cfg is babel's ("LaTeX Project Public License, either version
# 1.3 of this license or (at your option) any later version"), fonttext.cfg is LaTeX's
# (fontdef.dtx, "version 1.3c ... or any later version").
file hyphen.cfg lppl1.3
file fonttext.cfg lppl1.3c
# Unicode Character Database 14.0.0 (unicode-data): "For terms of use, see
# http://www.unicode.org/terms_of_use.html" (header). The database says lppl1.3c, which
# covers only the package's own load-unicode-data.tex.
file CaseFolding.txt unicode
file SpecialCasing.txt unicode
file UnicodeData.txt unicode
# pgf: "under the LaTeX Project Public License and/or under the GNU Public License" (header).
file pgfkeysfiltered.code.tex lppl1.3c
# Written by Tectonic for its own format file (MIT, the Tectonic licence).
file tectonic-format-latex.tex mit
